# Hermes – MCP Home Assistant Add-on

# AVISO IMPORTANTE:
# El paquete 'mcp' incluye FastMCP 1.0 como mcp.server.fastmcp.
# NO instalar el paquete independiente 'fastmcp' de PyPI, que es un
# proyecto distinto (FastMCP 2.x de Jeremiah Lowin / Prefect) con otra API.
# Mezclarlos rompe el código.

# ── SDK MCP ────────────────────────────────────────────────────
# Rama 2.x. Habla la revisión 2026-07-28 del protocolo de forma nativa, que es
# la que abren los clientes modernos: con 1.x recibían un 400 y caían al
# handshake clásico, gastando un round trip por conexión. El renombrado FastMCP → MCPServer y la reubicación de
# `stateless_http` / `transport_security` a `streamable_http_app()` ya están
# aplicados. Rango cerrado a 2.x: v3 volvería a poder romper el API.
mcp>=2,<3

# ── Core ───────────────────────────────────────────────────────
pydantic>=2,<3
aiohttp>=3.9,<4
structlog>=24,<27
pyyaml>=6,<7
uvicorn>=0.30,<1
starlette>=0.37,<2

# ── Auth ───────────────────────────────────────────────────────
# authlib para OAuth 2.1 AS+RS (RFC 6749, 7636, 7591, 7009).
# cryptography es dependencia transitiva de authlib para firma JWT.
authlib>=1.3,<2

# ── Utilidades ─────────────────────────────────────────────────
# psutil: resolución tailscale0, hassio bridge, crash loop detection.
# NO preinstalado en las imágenes base oficiales.
psutil>=5.9,<8

# packaging: comparación de versiones para smoke check de HA core.
packaging>=23,<27
